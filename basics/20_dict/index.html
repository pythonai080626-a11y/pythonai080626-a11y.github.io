# Basic dict — every key maps to one value person = {'name': 'Dana', 'age': 17, 'city': 'Haifa'} # An empty dict — {} IS a dict (not a set!) empty = {} # Duplicate keys: the LAST one wins d = {'a': 1, 'a': 2} # → {'a': 2} # Values can be anything — keys must be hashable mixed = {'nums': [1, 2], 7: True, (0, 0): 'origin'}

🏗️
dict() Constructor
From keywords, from pairs, or from two lists with zip()
02
# From keyword arguments (keys become strings) a = dict(name='Dana', age=17) # → {'name': 'Dana', 'age': 17} # From a list of (key, value) tuples b = dict([('x', 1), ('y', 2)]) # → {'x': 1, 'y': 2} # From two parallel lists — zip() pairs them up keys = ['red', 'green', 'blue'] values = ['#f00', '#0f0', '#00f'] colors = dict(zip(keys, values)) # → {'red': '#f00', 'green': '#0f0', 'blue': '#00f'}
🗝️
dict.fromkeys(keys, value)
Every key starts with the same value
03
# Same starting value for every key scores = dict.fromkeys(['a', 'b', 'c'], 0) # → {'a': 0, 'b': 0, 'c': 0} # No value given → every value is None blank = dict.fromkeys('abc') # → {'a': None, 'b': None, 'c': None} # ⚠️ All keys SHARE one mutable value! bad = dict.fromkeys(['a', 'b'], []) bad['a'].append(1) # → {'a': [1], 'b': [1]} both changed!
Fix for the trap: give each key its own list — start from {} and fill it in a loop: for k in keys: d[k] = []

.setdefault(key, default)
Get it — and create it first if it is missing
LIVE DEMO

in  ·  len()  ·  looping
Membership tests KEYS, not values — and it is instant
LIVE DEMO
d = {'a': 10, 'b': 20, 'c': 30} # `in` checks KEYS — O(1), no searching print('a' in d) # → True print(10 in d) # → False (10 is a value, not a key) print(10 in d.values()) # → True # len() = number of pairs print(len(d)) # Looping a dict loops its KEYS for k in d: print(k, d[k])

🧮
The Counting Pattern
The #1 real-world use of a dict — tally things up
LIVE DEMO
text = 'mississippi' counts = {} for ch in text: counts[ch] = counts.get(ch, 0) + 1 print(counts) # → {'m': 1, 'i': 4, 's': 4, 'p': 2}
>>>click ▶ RUN
Notice you cannot write counts[ch] += 1 here — on the first letter the key does not exist yet, so += 1 reads a missing key and raises KeyError. That is exactly the problem defaultdict solves ↓
🪄
defaultdict(int)
A dict that invents a starting value — so += just works
LIVE DEMO
from collections import defaultdict text = 'mississippi' # int() is 0 — every new key starts at 0 counts = defaultdict(int) for ch in text: counts[ch] += 1 # no .get(), no KeyError print(counts['s']) # → 4 # A letter that was never there is simply 0 print(counts['z']) # → 0 (and 'z' is now IN the dict!) # Turn it back into a plain dict when done plain = dict(counts)
>>>click ▶ RUN
The factory: you pass a function, not a value. defaultdict(int) → 0, defaultdict(list) → [], defaultdict(str) → '', defaultdict(set) → set(). Python calls it fresh for every missing key. Careful: just looking at a missing key creates it — use counts.get(ch) when you only want to peek, since .get() never adds anything.
