from collections import defaultdict
def char_count(word):

    return counts

The Fibonacci Sequence

for loop · building a list

כתבו פונקציה בשם fibonacci_list פרמטר אחד, בשם n היא מחזירה רשימה עם n המספרים הראשונים בסדרת פיבונאצ'י (כל מספר הוא סכום שני המספרים שלפניו: 0, 1, 1, 2, 3, 5, ...)

python program
def fibonacci_list(n):
    result = []
    a, b = 0, 1
    for _ in range(n):
        result.append(a)
        a, b = b, a + b
    return result


print(fibonacci_list(6))
expected output
[0, 1, 1, 2, 3, 5]

The Second Largest

for loop · tracking two values

כתבו פונקציה בשם second_largest פרמטר אחד, בשם numbers - רשימה של מספרים שונים זה מזה היא מחזירה את המספר השני בגודלו ברשימה, בלי למיין אותה עברו על הרשימה פעם אחת, ושמרו את הגדול והשני בגודלו תוך כדי

python program
def second_largest(numbers):
    first = second = None
    for n in numbers:
        if first is None or n > first:
            second = first
            first = n
        elif n != first and (second is None or n > second):
            second = n
    return second


print(second_largest([4, 1, 9, 2]))
expected output
4

Group By Word Length

for loop · dict of lists

כתבו פונקציה בשם group_by_length פרמטר אחד, בשם words - רשימה של מילים היא מחזירה מילון: כל אורך מילה הוא מפתח, והערך הוא רשימה של כל המילים באורך הזה עבור כל מילה - אם האורך שלה עוד לא מפתח במילון, התחילו בשבילו רשימה ריקה, ואז הוסיפו אליה את המילה אפשר גם עם dict רגיל, ואפשר גם עם defaultdict שניתן למעלה

python program
from collections import defaultdict
def group_by_length(words):
    groups = {}
    for w in words:
        n = len(w)
        if n not in groups:
            groups[n] = []
        groups[n].append(w)
    return groups


print(group_by_length(['a', 'bb', 'cc', 'ddd', 'e']))
expected output
{1: ['a', 'e'], 2: ['bb', 'cc'], 3: ['ddd']}

Safe Nested Lookup

dict of dicts · get

כתבו פונקציה בשם nested_get שלושה פרמטרים: d (מילון של מילונים), outer_key ו-inner_key היא מחזירה את d[outer_key][inner_key] אבל אם outer_key לא קיים ב-d, או ש-inner_key לא קיים במילון הפנימי - מחזירה None במקום לקרוס השתמשו ב-get במקום בסוגריים מרובעים

python program
def nested_get(d, outer_key, inner_key):
    inner = d.get(outer_key)
    if inner is None:
        return None
    return inner.get(inner_key)


print(nested_get({'a': {'x': 1}}, 'a', 'x'))
print(nested_get({'a': {'x': 1}}, 'z', 'x'))
expected output
1
None

Extract The Numbers

regex · re.findall

כתבו פונקציה בשם extract_numbers פרמטר אחד, בשם text יש לכם למעלה pattern מוכן בשם NUMBER_PATTERN, שכבר יודע לזהות רצף של ספרות בתוך טקסט - לא צריך לכתוב regex בעצמכם השתמשו ב-re.findall עם ה-pattern הזה כדי למצוא את כל רצפי הספרות בטקסט, והחזירו אותם כרשימה של מספרים שלמים (לא כמחרוזות - צריך להמיר כל אחד עם int)

python program
import re

# a ready-made pattern - matches one or more digits in a row
NUMBER_PATTERN = r'\d+'
def extract_numbers(text):
    matches = re.findall(NUMBER_PATTERN, text)
    return [int(m) for m in matches]


print(extract_numbers('I have 3 cats and 12 dogs, and 007 fish'))
expected output
[3, 12, 7]
