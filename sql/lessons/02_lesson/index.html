1) Create & Drop Table

CREATE TABLE shopping (id INTEGER PRIMARY KEY, name TEXT, amount INTEGER); DROP TABLE shopping
התוצאה אחרי CREATE TABLE:
idnameamount
(0 שורות — הטבלה קיימת, אבל ריקה)

CREATE TABLE מגדירה את מבנה הטבלה: שלוש עמודות, כאשר id הוא המפתח הראשי. DROP TABLE היא הפעולה ההפוכה — היא מוחקת את הטבלה ואת כל מה שבתוכה, כולל המבנה עצמו. היא מוצגת כאן בשביל התחביר בלבד, אבל היא לא באמת מורצת בסיור הזה — אנחנו משאירים את shopping בחיים לכל שלב שבא אחריה.

2) Rename a Table

ALTER TABLE shopping RENAME TO shopp; ALTER TABLE shopp RENAME TO shopping;

(0 שורות — עדיין ריקה, ועדיין נקראת shopping)

ALTER TABLE ... RENAME TO משנה את שם הטבלה בלי לגעת בנתונים שבתוכה. כאן היא משנה שם פעמיים ברצף — ל-shopp וחזרה מיד ל-shopping — כך שאחרי שתי השורות הטבלה נמצאת בדיוק באותו מקום שבו התחילה, רק כדי להראות שהתחביר עובד בשני הכיוונים.

20) JOIN + Subquery (MAX)

SELECT s.id, s.name, s.amount, s.maavar, p.price FROM shopping s JOIN prices p ON s.id = p.id WHERE p.price = (SELECT MAX(price) FROM prices);
