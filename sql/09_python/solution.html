db.py

The Lesson 9 template. Only DB_PATH changed.

db.py
import sqlite3

DB_PATH = "movies.db"

def connect():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def run_query_select(sql, params=()):
    conn = connect()
    try:
        cursor = conn.cursor()
        cursor.execute(sql, params)
        rows = cursor.fetchall()
        return [dict(row) for row in rows]
    finally:
        conn.close()

def run_update_query(sql, params=()):
    conn = connect()
    try:
        cursor = conn.cursor()
        cursor.execute(sql, params)
        conn.commit()
        return cursor.lastrowid
    finally:
        conn.close()

movies_dal.py

Five functions, one query each. This is the only file with SQL in it.

movies_dal.py
from db import run_query_select, run_update_query


def get_all_movies():
    sql = "SELECT * FROM movies"
    return run_query_select(sql)


def get_movies_by_year(year):
    sql = "SELECT * FROM movies WHERE year = ?"
    return run_query_select(sql, (year,))


def update_movie_title(movie_id, new_title):
    sql = "UPDATE movies SET title = ? WHERE id = ?"
    run_update_query(sql, (new_title, movie_id))


def delete_movie(movie_id):
    sql = "DELETE FROM movies WHERE id = ?"
    run_update_query(sql, (movie_id,))


def insert_movie(title, director, year, genre, rating):
    sql = """INSERT INTO movies (title, director, year, genre, rating)
             VALUES (?, ?, ?, ?, ?)"""
    run_update_query(sql, (title, director, year, genre, rating))
  • SELECT uses run_query_select and returns the rows. UPDATE, DELETE and INSERT use run_update_query.
  • Every value from the user goes in through a ?, never pasted into the SQL text.
  • The order of the values in the tuple matches the order of the ? marks: in UPDATE, the title comes first, then the id.

main.py

Two helper functions, then one while True loop that runs until the user picks 6.

main.py
from movies_dal import (get_all_movies, get_movies_by_year,
                        update_movie_title, delete_movie, insert_movie)


def show_menu():
    print()
    print("===== MOVIES MENU =====")
    print("1. Show all movies")
    print("2. Find movies by year")
    print("3. Update movie title")
    print("4. Delete movie")
    print("5. Add new movie")
    print("6. Exit")


def print_movies(movies):
    for movie in movies:
        print(movie["id"], "|", movie["title"], "|", movie["director"], "|",
              movie["year"], "|", movie["genre"], "|", movie["rating"])


while True:
    show_menu()
    choice = input("Choose an option: ")

    if choice == "1":
        print_movies(get_all_movies())

    elif choice == "2":
        year = int(input("Enter year: "))
        print_movies(get_movies_by_year(year))

    elif choice == "3":
        movie_id = int(input("Enter movie id: "))
        new_title = input("Enter new title: ")
        update_movie_title(movie_id, new_title)
        print("Movie updated!")

    elif choice == "4":
        movie_id = int(input("Enter movie id: "))
        delete_movie(movie_id)
        print("Movie deleted!")

    elif choice == "5":
        title = input("Title: ")
        director = input("Director: ")
        year = int(input("Year: "))
        genre = input("Genre: ")
        rating = float(input("Rating: "))
        insert_movie(title, director, year, genre, rating)
        print("Movie added!")

    elif choice == "6":
        print("Bye!")
        break

    else:
        print("Invalid option")
  • input() returns text, so the choice is compared to "1", not 1.
  • Year and id become numbers with int(), the rating with float().
  • break on option 6 is the only way out of the loop.
